# Prueba de humo del modo sin ventana y del depurador (entregables 3, 5 y 6).
# Se ejecuta con:  umkcli --rom rom/monitor.bin --script tests/cli/humo.txt

# El monitor arranca y muestra el "-" de listo.
run 400000
display

# Cargar el programa del criterio 2 tecleando en el panel real.
keys P,0,8,0,0,SPACE,3,E,SPACE,A,A,SPACE,C,3,SPACE,0,0,SPACE,0,8,SPACE,VP
mem 0800 8
dis 0800 2

# Punto de ruptura en el segundo byte del programa de usuario.
bp 0802
keys ST,0,8,0,0,VP
run 200000
regs
panel

# Guardar y restaurar el estado completo.
state save build/humo_estado.bin
reg A 00
state load build/humo_estado.bin
regs

# Intel HEX de ida y vuelta.
savehex build/humo.hex 0800 8
poke 0800 00 00 00 00 00
loadhex build/humo.hex
mem 0800 8

# Paso por ciclo de máquina desde el panel.
bp clear
sw step on
sw cycle on
cycle 4
quit
